cmake_minimum_required(VERSION 3.16)
project(FleetControlConnectionTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Network Quick Test)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Network Quick Test)
add_executable(fleetcheck ../fleetcontrol_connection_test.cpp ../../app/fleetcontrol/fleetcontrollte.cpp ../../app/fleetcontrol/fleetcontrollte.h)
target_link_libraries(fleetcheck PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Network Qt${QT_VERSION_MAJOR}::Quick Qt${QT_VERSION_MAJOR}::Test)
enable_testing()
add_test(NAME fleet_connection COMMAND fleetcheck ${CMAKE_CURRENT_SOURCE_DIR}/../../qml/ui/configpopup/openhd_settings/FleetControlSettingsPanel.qml)
set_tests_properties(fleet_connection PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QUICK_CONTROLS_STYLE=Material")
